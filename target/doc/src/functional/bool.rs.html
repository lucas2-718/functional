3/// Basic data describing the booleans
4#[derive(Clone,Debug)]
5pub struct BoolData {
6    /// The trimming function used internally, returns 0 if the input is zero or 1 and otherwise returns 1
7    pub trim: FinalTerm,
8    /// The type of booleans, a sum requiring trim to be zero
9    pub bool_type: FinalTerm,
10    /// The false boolean, of type bool_type
11    pub bool_false: FinalTerm,
12    /// The true boolean, of type bool_type
13    pub bool_true: FinalTerm,
14    /// The family used in the [Sig] of [BoolData::bool_type]
15    pub bool_fam: FinalTerm,
16}
17
18impl Scopeless for BoolData {}

20impl BoolData {
21    /// Creates a new BoolData instance
22    pub fn new() -> Res<BoolData> {
23        // Construct the function trim, which returns 0 if the input is zero or one, and 1 otherwise by repeated pattern matching
24        
25        let family = Lam(Nat.ctn()?,Nat.ctn()?,"_".into()).ctn()?;
26        let trim = lam_helper([family], Nat.ctn()?, "n", |[family],n|{
27            App(App(App(NatInd(n, 0u8.into()).ctn()?,family.clone()).ctn()?,Zero.ctn()?).ctn()?,lam_helper([family], Nat.ctn()?, "n-1", |[family],n|{
28                lam_helper([family,n],Nat.ctn()?,"_",|[family,n],_|{
29                    App(App(App(NatInd(n, 0u8.into()).ctn()?,family.clone()).ctn()?,Zero.ctn()?).ctn()?,Lam(Nat.ctn()?,Lam(Nat.ctn()?,Succ(Zero.ctn()?).ctn()?,"_".into()).ctn()?,"n2".into()).ctn()?).ctn()
30                })
31            })?).ctn()
32        })?;
33
34        let bfam = lam_helper([trim.clone()], Nat.ctn()?, "n", |[trim],n|{
35            straight_eq(Zero.ctn()?, App(trim,n).ctn()?)
36        })?;
37
38        let bool_type = Sig(bfam.clone()).ctn()?.fin()?;
39
40        let bool_false = Pair(bfam.clone(), Zero.ctn()?, refl(Zero.ctn()?)?).ctn()?.fin()?;
41
42        let bool_true = Pair(bfam.clone(), Succ(Zero.ctn()?).ctn()?, refl(Zero.ctn()?)?).ctn()?.fin()?;
43
44        let trim = trim.fin()?;
45
46        Ok(BoolData { trim, bool_type, bool_false, bool_true, bool_fam: bfam.fin()? })
47    }
48    /// Runs the boolean induction principle on a specific instance of a boolean
49    pub fn bool_ind(&self, b: ContainedTerm, n: Natural) -> Res<ContainedTerm> {
50        let famtype = pi_helper_poly([], self.bool_type.get(), "_", [n], |[],_,[n]|{Universe(n).ctn()})?;
51        let ind = lam_helper_poly([b,self.trim.get(),self.bool_false.get(),self.bool_true.get(),self.bool_fam.get()], famtype, "fam", [n], |[b,trim,bfalse,btrue,bfam],fam,[n]|{
52            let true_family = lam_helper([fam.clone(),bfalse.clone(),btrue.clone()], Sig(bfam.clone()).ctn()?, "b", |[fam,bfalse,btrue],b|{
53                pi_helper([btrue,fam.clone(),b], App(fam,bfalse).ctn()?, "vf", |[btrue,fam,b],_|{
54                    pi_helper([fam.clone(),b], App(fam,btrue).ctn()?, "vt", |[fam,b],_|{
55                        App(fam,b).ctn()
56                    })
57                })
58            })?;
59            App(App(SigInd(b, n).ctn()?,true_family.clone()).ctn()?,lam_helper_poly([trim,fam,bfalse,btrue,bfam], Nat.ctn()?, "n", [n],|[trim,fam,bfalse,btrue,bfam],n,[u]|{
60                // Goal : (hyp: 0 = trim n) -> fam (0, refl 0) -> fam (1, refl 0) -> fam (n, hyp)
61                // how to prove goal
62                // matching on goal
63                // (hyp : 0 = 0) -> (vf: fam (0, refl 0)) -> (vt: fam (1, refl 0)) : fam (0, hyp) := transport vt over refl 0 = hyp
64                // (hyp : 0 = 0) -> (vf: fam (0, refl 0)) -> (vt: fam (1, refl 0)) : fam (1, hyp) := transport vt over refl 0 = hyp
65                // (hyp : 0 = 1) : (vf: fam (0, refl 0)) -> (vt: fam (1, refl 0)) -> fam (2 + n, hyp) := exfalso into the family
66                let indfam = lam_helper([bfalse.clone(),btrue.clone(),fam.clone(),trim,bfam.clone()], Nat.ctn()?, "n", |[bfalse,btrue,fam,trim,bfam],n|{
67                    pi_helper([n.clone(),bfalse,btrue,fam,bfam], straight_eq(Zero.ctn()?,App(trim,n).ctn()?)?, "hyp", |[n,bfalse,btrue,fam,bfam],hyp|{
68                        pi_helper([n.clone(),btrue,fam.clone(),bfam,hyp],App(fam,bfalse).ctn()?,"vf",|[n,btrue,fam,bfam,hyp],_|{
69                            pi_helper([n.clone(),fam.clone(),bfam,hyp], App(fam,btrue).ctn()?, "vt", |[n,fam,bfam,hyp],_|{
70                                App(fam,Pair(bfam,n,hyp).ctn()?).ctn()
71                            })
72                        })
73                    })
74                })?;
75                
76                let fam0 = lam_helper([bfalse.clone(),btrue.clone(),bfam.clone(),fam.clone()],straight_eq(Zero.ctn()?, Zero.ctn()?)?,"hyp",|[bfalse,btrue,bfam,fam],hyp|{
77                    lam_helper([btrue,bfam,fam.clone(),hyp],App(fam,bfalse).ctn()?,"vf",|[btrue,bfam,fam,hyp],vf|{
78                        lam_helper([bfam,fam.clone(),hyp,vf], App(fam,btrue).ctn()?, "vt", |[bfam,fam,hyp,vf],_|{
79                            // goal = fam (0, hyp)
80                            // have = fam (0, refl 0)
81                            // refl = hyp
82                            let refl_hyp = App(numbers::zero_eq_trivial()?.get(),hyp).ctn()?;
83                            // h => fam (0 , h)
84                            let cong_func = lam_helper([fam,bfam],straight_eq(Zero.ctn()?, Zero.ctn()?)?,"hyp",|[fam,bfam],hyp|{
85                                App(fam,Pair(bfam,Zero.ctn()?,hyp).ctn()?).ctn()
86                            })?;
87                            // have = goal
88                            let path = cong(cong_func, refl_hyp)?;
89                            transport_eq(path,vf)
90                        })
91                    })
92                })?;
93
94                let fam1 = lam_helper([bfalse.clone(),btrue.clone(),bfam.clone(),fam.clone()],straight_eq(Zero.ctn()?, Zero.ctn()?)?,"hyp",|[bfalse,btrue,bfam,fam],hyp|{
95                    lam_helper([btrue,bfam,fam.clone(),hyp],App(fam,bfalse).ctn()?,"vf",|[btrue,bfam,fam,hyp],_|{
96                        lam_helper([bfam,fam.clone(),hyp], App(fam,btrue).ctn()?, "vt", |[bfam,fam,hyp],vt|{
97                            // goal = fam (1, hyp)
98                            // have = fam (1, refl 0)
99                            // refl = hyp
100                            let refl_hyp = App(numbers::zero_eq_trivial()?.get(),hyp).ctn()?;
101                            // h => fam (0 , h)
102                            let cong_func = lam_helper([fam,bfam],straight_eq(Zero.ctn()?, Zero.ctn()?)?,"hyp",|[fam,bfam],hyp|{
103                                App(fam,Pair(bfam,Succ(Zero.ctn()?).ctn()?,hyp).ctn()?).ctn()
104                            })?;
105                            // have = goal
106                            let path = cong(cong_func, refl_hyp)?;
107                            transport_eq(path,vt)
108                        })
109                    })
110                })?;
111
112                let fam2 = lam_helper([bfalse,btrue,bfam,fam.clone()],Nat.ctn()?,"n2",|[bfalse,btrue,bfam,fam],n2|{
113                    lam_helper([bfalse,btrue,bfam,fam,n2], straight_eq(Zero.ctn()?, Succ(Zero.ctn()?).ctn()?)?, "hyp", |[bfalse,btrue,bfam,fam,n2],hyp|{
114                        let n = Succ(Succ(n2).ctn()?).ctn()?;
115                        let target = pi_helper([n.clone(),btrue,fam.clone(),bfam,hyp.clone()],App(fam,bfalse).ctn()?,"vf",|[n,btrue,fam,bfam,hyp],_|{
116                            pi_helper([n.clone(),fam.clone(),bfam,hyp], App(fam,btrue).ctn()?, "vt", |[n,fam,bfam,hyp],_|{
117                                App(fam,Pair(bfam,n,hyp).ctn()?).ctn()
118                            })
119                        })?;
120                        App(FalseData::new()?.exfalso(target)?,hyp).ctn()
121                    })
122                })?;
123
124
125                let partial = App(App(App(NatInd(n, u).ctn()?,indfam.clone()).ctn()?,fam0).ctn()?,lam_helper_poly([fam1,fam2,indfam], Nat.ctn()?, "n1", [u],|[fam1,fam2,indfam],n1,[u]|{
126                    let new_indfam = lam_helper([indfam.clone()], Nat.ctn()?, "n",|[indfam],n|{
127                        App(indfam,Succ(n).ctn()?).ctn()
128                    })?;
129                    lam_helper_poly([fam1,fam2,new_indfam.clone(),n1.clone()], App(indfam,n1).ctn()?, "_", [u],|[fam1,fam2,indfam,n1],_,[u]|{
130                        App(App(App(NatInd(n1, u).ctn()?,indfam.clone()).ctn()?,fam1).ctn()?,lam_helper([fam2,indfam], Nat.ctn()?, "n2", |[fam2,indfam],n2|{
131                            lam_helper([fam2,n2.clone()],App(indfam,n2).ctn()?,"_",|[fam2,n2],_|{
132                                App(fam2,n2).ctn()
133                            })
134                        })?).ctn()
135                    })
136                })?).ctn()?;
137                
138                Ok(partial)
139            })?).ctn()
140        })?;
141        
142        Ok(ind)
143    }
144    /// Creates the boolean induction principle on a specific universe level
145    pub fn generic_bool_ind(&self, n: Natural) -> Res<FinalTerm> {
146        lam_helper_poly([], self.bool_type.get(), "b", (self.clone(),n), |[],b,(this,n)|{
147            this.bool_ind(b, n)
148        })?.fin()
149    }
