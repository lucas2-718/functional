3/// Basic data about the type with no inhabitants, False
4/// 
5/// False is internally represented by 0 = 1, so any value of that may be used as a value of false
6#[derive(Clone)]
7pub struct FalseData {
8    /// The False type that has no inhabitants and indicates logical impossibility
9    pub false_type: FinalTerm,
10}
11
12impl Scopeless for FalseData {}
13
14impl FalseData {
15    /// Create an instance of FalseData
16    pub fn new() -> Res<Self> {
17        Ok(Self {
18            false_type : straight_eq(Zero.ctn()?,Succ(Zero.ctn()?).ctn()?)?.fin()?
19        })
20    }
21    /// Given a logically impossible term, make a term of any type via transport
22    pub fn exfalso(&self, target: ContainedTerm) -> Res<ContainedTerm> {
23        match target.clone().typed()?.pop() {
24            Universe(n) => {
25                let other: ContainedTerm = if n==0u8.into() {
26                    Zero.ctn()?
27                } else {
28                    if n==1u8.into() {
29                        Nat.ctn()?
30                    } else {
31                        Universe(n.pred().pred()).ctn()?
32                    }
33                };
34                let othertype = other.clone().typed()?;
35                let exfalso: ContainedTerm = lam_helper_poly([othertype,target,other], self.false_type.get(), "hyp", [n],|[othertype,target,other],hyp,[n]|{
36                    // Using the false equality, construct an equality between two types
37                    let family = lam_helper_poly([othertype,target,hyp], II.ctn()?, "i", [n],|[othertype,target,hyp],i,[n]|{
38                        let number = EqUw(hyp, i).ctn()?;
39                        App(App(App(NatInd(number, n.suc()).ctn()?,Lam(Nat.ctn()?,Universe(n).ctn()?,"_".into()).ctn()?).ctn()?,othertype).ctn()?,lam_helper_poly([target], Nat.ctn()?, "_", [n],|[target],_,[n]|{
40                            lam_helper([target], Universe(n).ctn()?, "_", |[target],_|{
41                                Ok(target)
42                            })
43                        })?).ctn()
44                    })?;
45
46                    // then transport a known value in one type to the other arbitrary type, completing the exfalso
47                    let transport = Transp(family, other).ctn()?;
48                    
49                    Ok(transport)
50                })?;
51                
52                Ok(exfalso)
53            }
54            _ => Err(ErrorType::new("target is not a type!".into()))
55        }
56    }
57}
