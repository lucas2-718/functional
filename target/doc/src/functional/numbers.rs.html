11/// A function where 0 => 0 and _ => 1
12/// 
13/// Helpful for converting 0 = succ x to 0 = 1, which is what the false type is internally
14pub fn flatten_natural(n: ContainedTerm) -> Res<ContainedTerm> {
15    let fam = Pi(Nat.ctn()?,Nat.ctn()?,"_".into()).ctn()?;
16    App(App(App(NatInd(n, 0u8.into()).ctn()?,fam).ctn()?,Zero.ctn()?).ctn()?,Lam(Nat.ctn()?, Lam(Nat.ctn()?, Succ(Zero.ctn()?).ctn()?, "_".into()).ctn()?, "_".into()).ctn()?).ctn()
17}

19/// Proof that all values of 0=0 are in fact refl
20/// 
21/// produces a term typed (h : 0 = 0) -> refl 0 = h
22pub fn zero_eq_trivial() -> Res<FinalTerm> {
23    // Proving that (h : 0=0) -> refl 0 = h is relatively trivial
24    // we can prove that h i = 0 via casing on what h i is and then path splitting
25    // this ensures the endpoint behavior being that h i = 0 is refl and not some variant of h
26    // then reassemble along the path to get the full value
27    lam_helper([], straight_eq(Zero.ctn()?, Zero.ctn()?)?, "hyp", |[],hyp|{
28        // izero : (i : II) -> 0 = hyp i
29        // izero i0 _ = refl 0
30        // izero i1 _ = refl 0
31        // izero _ i0 = refl 0
32        // izero _ i1 = hyp
33        
34        let izero = lam_helper([hyp], II.ctn()?, "i", |[hyp],i|{
35            // (n : nat) -> (0 = n) -> (0 = n)
36            // make an equality into a reflexive equality
37            // but a different variant of this
38            let fam = lam_helper([], Nat.ctn()?, "n", |[],n|{
39                pi_helper([n.clone()], straight_eq(Zero.ctn()?, n)?, "_", |[n],_|{
40                    straight_eq(Zero.ctn()?, n)
41                })
42            })?;
43
44            // Case on the natural to ensure definitional computation on zero works correctly and then just do whatever works for the impossible case
45            let func = lam_helper([fam], Nat.ctn()?, "n", |[fam],n|{
46                App(App(App(NatInd(n, 0u8.into()).ctn()?,fam.clone()).ctn()?,
47                    lam_helper([], straight_eq(Zero.ctn()?, Zero.ctn()?)?, "_", |[],_|{refl(Zero.ctn()?)})?).ctn()?,
48                    lam_helper([fam], Nat.ctn()?, "n", |[fam],n|{
49                        lam_helper([n.clone()],App(fam,n).ctn()?, "_", |[n],_|{
50                            // This case is impossible, so doesn't matter as long as it type-checks
51                            lam_helper([], straight_eq(Zero.ctn()?, Succ(n).ctn()?)?, "hyp", |[],hyp|{Ok(hyp)})
52                        })
53                    })?).ctn()
54            })?;
55
56            let split = split_path(hyp.clone(), IA.ctn()?, i.clone())?;
57
58            // logically the same as split, but now reflexive whenever it can be
59            let new_split = App(App(func,EqUw(hyp,i).ctn()?).ctn()?,split).ctn()?;
60
61            Ok(new_split)
62        })?;
63
64        
65        
66        EqLam(lam_helper([izero], II.ctn()?, "i", |[izero],i|{
67            EqLam(lam_helper([izero,i], II.ctn()?, "j", |[izero,i],j|{
68                EqUw(App(izero,j).ctn()?,i).ctn()
69            })?).ctn()
70        })?).ctn()
71        
72    })?.fin()
73}

76/// A struct with some proofs and functions about [Nat]
77#[derive(Clone,Debug)]
78pub struct NatData {
79    /// The function that does addition
80    /// 
81    /// Works about how you would expect, but only computes on first argument
82    /// 
83    /// Use [NatData::add_zero_right] and [NatData::add_succ_right] to do things as if it computes on the second argument
84    /// 
85    /// Alternately use [NatData::add_sym], but that involves concatenated paths, which might cause problems 
86    pub add_func: FinalTerm,
87    /// Proof that (x : nat) -> add x 0 = x
88    /// 
89    /// Analogue to left computation rule on zero
90    pub add_zero_right: FinalTerm,
91    /// Proof that (x y : nat) -> add x (S y) = S (add x y)
92    /// 
93    /// Analogue to left computation rule on successor
94    pub add_succ_right: FinalTerm,
95    /// Proof that (x y : nat) -> add x y = add y x
96    /// 
97    /// Symmetric property of addition
98    pub add_sym: FinalTerm
99}
100
101impl Scopeless for NatData {}
102
103impl NatData {
104    /// Creates a new instance of NatData with the functions and proofs inside the struct
105    pub fn new() -> Res<Self> {
106        let addition = lam_helper([], Nat.ctn()?, "n", |[],n|{
107            let family = Lam(Nat.ctn()?,Pi(Nat.ctn()?,Nat.ctn()?,"x".into()).ctn()?,"n".into()).ctn()?;
108            let base = lam_helper([], Nat.ctn()?, "x", |[],x|{Ok(x)})?;
109            let step = lam_helper([],Nat.ctn()?,"n",|[],_|{
110                lam_helper([], Pi(Nat.ctn()?,Nat.ctn()?,"n".into()).ctn()?, "f", |[],f|{
111                    lam_helper([f], Nat.ctn()?, "x", |[f],x|{
112                        // add (S n) ? = S (add n ?)
113                        // there is an equivalent form of
114                        // add (S n) ? = add n (S ?)
115                        // but it is harder to work with definitionally
116                        Succ(App(f,x).ctn()?).ctn()
117                    })
118                })
119            })?;
120            App(App(App(NatInd(n, 0u8.into()).ctn()?,family).ctn()?,base).ctn()?,step).ctn()
121        })?;
122        
123        // add x 0 = x
124        // add 0 0 = 0 -> refl
125        // add (S x) 0 = S x -> cong S (prev : add x 0 = x)
126        let add_zero_right = lam_helper([addition.clone()], Nat.ctn()?, "x", |[addition],x|{
127            let family = lam_helper([addition.clone()], Nat.ctn()?, "x", |[addition],x|{
128                straight_eq(App(App(addition,x.clone()).ctn()?,Zero.ctn()?).ctn()?, x)
129            })?;
130            let base = refl(Zero.ctn()?)?;
131            let step = lam_helper([addition], Nat.ctn()?, "x", |[addition],x|{
132                lam_helper([], straight_eq(App(App(addition,x.clone()).ctn()?,Zero.ctn()?).ctn()?, x)?, "p", |[],p|{
133                    cong(successor_function()?, p)
134                })
135            })?;
136
137            App(App(App(NatInd(x, 0u8.into()).ctn()?,family).ctn()?,base).ctn()?,step).ctn()
138        })?;
139
140        // add x (S y) = S (add x y)
141        // add 0 (S y) = S y -> refl
142        // add (S x) (S y) = S (add (S x) y)
143        // ~ S (add x (S y)) = S (S (add x y)) -> cong S (prev : add x (S y) = S (add x y))
144        let add_succ_right = lam_helper([addition.clone()], Nat.ctn()?, "x", |[addition],x|{
145            lam_helper([addition,x], Nat.ctn()?, "y", |[addition,x],y|{
146                let family = lam_helper([addition.clone(),y.clone()],Nat.ctn()?,"x",|[addition,y],x|{
147                    straight_eq(App(App(addition.clone(),x.clone()).ctn()?,Succ(y.clone()).ctn()?).ctn()?, Succ(App(App(addition,x).ctn()?,y).ctn()?).ctn()?)
148                })?;
149                let base = refl(Succ(y.clone()).ctn()?)?;
150                let step = lam_helper([family.clone()],Nat.ctn()?,"x",|[family],x|{
151                    lam_helper([],App(family,x).ctn()?,"p",|[],p|{
152                        cong(successor_function()?,p)
153                    })
154                })?;
155
156                App(App(App(NatInd(x, 0u8.into()).ctn()?,family).ctn()?,base).ctn()?,step).ctn()
157            })
158        })?;
159
160        // add x y = add y x
161
162        // add x (S y) = add (S y) x
163        // ~ add x (S y) = S (add y x)
164        // -> asr x y . (? : (add x y = add y x) = cong S prev)
165        
166        // add x 0 = add 0 x
167        // add x 0 = x -> azr x
168        let add_sym = lam_helper([addition.clone(),add_zero_right.clone(),add_succ_right.clone()],Nat.ctn()?,"x",|[addition,azr,asr],x|{
169            lam_helper([addition,x,azr,asr], Nat.ctn()?, "y", |[addition,x,azr,asr],y|{
170                let family = lam_helper([addition.clone(),x.clone()], Nat.ctn()?, "y", |[addition,x],y|{
171                    straight_eq(App(App(addition.clone(),x.clone()).ctn()?,y.clone()).ctn()?, App(App(addition,y).ctn()?,x).ctn()?)
172                })?;
173                let base = App(azr,x.clone()).ctn()?;
174                let step = lam_helper([x,asr,family.clone()], Nat.ctn()?, "y", |[x,asr,fam],y|{
175                    lam_helper([x,y.clone(),asr], App(fam,y).ctn()?, "prev", |[x,y,asr],prev|{
176                        trans(App(App(asr,x).ctn()?,y).ctn()?,cong(successor_function()?,prev)?)
177                    })
178                })?;
179                
180                App(App(App(NatInd(y, 0u8.into()).ctn()?,family).ctn()?,base).ctn()?,step).ctn()
181            })
182        })?;
183
184        let add_func = addition.fin()?;
185        let add_zero_right = add_zero_right.fin()?;
186        let add_succ_right = add_succ_right.fin()?;
187        let add_sym = add_sym.fin()?;
188
189        Ok(NatData { add_func, add_zero_right, add_succ_right, add_sym })
190    }
